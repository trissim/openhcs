\section{Finite Basis for the Current Positive Landscape}

We distinguish between \emph{tractable families treated here} and \emph{primitive tractability mechanisms}. A tractable family is any named positive condition included in the present analysis. A primitive mechanism is a smaller conceptual source of tractability that may explain several families at once. The positive theory organizes into three roles: six core structural families, four regime lifts, and five degenerate collapses. Projecting those families to primitive mechanisms leaves eight sources of tractability: the six core subcases together with constant-optimizer collapse and finite explicit enumeration.

\leanmetapending{\LHrng{FR}{1}{2}}
\begin{theorem}[Explicit Partition of the Current Positive Landscape]\label{thm:current-landscape-partition}
The fifteen tractable families considered here admit an explicit disjoint partition into three blocks:
\begin{enumerate}
\item six \emph{core} structural families,
\item four \emph{lifted} families that reduce to existing core subcases, and
\item five \emph{degenerate} families that become tractable by optimizer collapse or finite explicit enumeration.
\end{enumerate}
Moreover, each family is classified into exactly one of these three blocks by an explicit classification map.
\end{theorem}

\leanmetapending{\LHrng{FR}{25}{28}}
\begin{proposition}[Explicit Inventory]\label{prop:explicit-inventory}
The fifteen tractable families considered here are exactly the following:
\begin{enumerate}
\item core: bounded actions, separable utility, low tensor rank, tree structure, bounded treewidth, coordinate symmetry;
\item lifts: product distribution, bounded support, bounded horizon, full observability;
\item degenerate: single action, bounded state space, strict global dominance, constant optimal set, multiplicative-separable constant-sign.
\end{enumerate}
\end{proposition}

\leanmetapending{\LH{FR3}}
\begin{corollary}[Finite Basis for the Current Positive Landscape]\label{cor:finite-basis-current}
Projecting the partition of Theorem~\ref{thm:current-landscape-partition} to primitive mechanisms yields an explicit finite list of eight primitive tractability mechanisms. The list consists of the six core structural subcases
\[
\text{bounded actions},\ 
\text{separable utility},\ 
\text{low tensor rank},\ 
\text{tree structure},\ 
\text{bounded treewidth},\ 
\text{coordinate symmetry},
\]
together with two degenerate mechanisms:
\[
\text{constant-optimizer collapse}
\qquad\text{and}\qquad
\text{finite explicit enumeration}.
\]
\end{corollary}

\leanmetapending{\LH{FR29}}
\begin{proposition}[Explicit Primitive Basis]\label{prop:explicit-basis}
The current primitive basis consists exactly of bounded actions, separable utility, low tensor rank, tree structure, bounded treewidth, coordinate symmetry, constant-optimizer collapse, and finite explicit enumeration.
\end{proposition}

The current positive landscape therefore contains fifteen tractable families accounted for by eight primitive mechanisms.

\leanmetapending{\LH{FR101}, \LHrng{FR}{105}{106}, \LH{FR108}, \LHrng{FR}{112}{113}}
\begin{proposition}[All Independent Core Mechanisms Are Nondegenerate]\label{prop:four-core-nondegenerate}
The analysis already contains witness families showing that five core mechanisms are individually indispensable in the sense of basis coverage:
\begin{enumerate}
\item a low-rank-only witness family,
\item a separable-only witness family,
\item a bounded-actions-only witness family,
\item a bounded-treewidth-only witness family, and
\item a symmetry-only witness family.
\end{enumerate}
Each of these families remains tractable by its designated mechanism while failing the corresponding versions of the other independent core mechanisms.
\end{proposition}

Proposition~\ref{prop:four-core-nondegenerate} settles the independent part of the basis question on the present surface. Every core mechanism that is not merely contained in another is shown to be genuinely needed.

Concretely, the present theory contains five kinds of ``only-by'' witness: low-rank only, separable only, bounded-actions only, bounded-treewidth only, and symmetry only.

\leanmetapending{\LH{FR114}}
\begin{proposition}[Weak Tree Ordering Does Not Imply Width One]\label{prop:legacy-tree-gap}
There exists a dependency presentation satisfying the older monotone tree-order predicate \texttt{TreeStructured} whose dependency graph has real treewidth greater than one. Width-one containment requires the stricter parent-tree condition from Proposition~\ref{prop:parent-tree-width-one}, not merely monotone ordering.
\end{proposition}

\leanmetapending{\LH{FR115}}
\begin{proposition}[Cyclic Dependencies Recover Hardness]\label{prop:tree-hardness-bridge}
The reduction family for exact relevance certification yields \coNP-hard instances with cyclic dependency structure. Thus the tree-structured regime is not only sufficient for tractability; leaving it also reaches the hard side.
\end{proposition}

\leanmetapending{\LH{FR100}, \LHrng{FR}{110}{111}}
\begin{proposition}[Parent-Tree Structure Gives Width-One Decompositions]\label{prop:parent-tree-width-one}
Suppose a dependency presentation is tree-structured in the strict parent sense: every dependency points to a smaller coordinate and every coordinate has at most one parent. Then the induced dependency graph admits a genuine tree decomposition of width at most one.
\end{proposition}

Proposition~\ref{prop:parent-tree-width-one} is the containment theorem behind the intended ``tree structure is the treewidth-one special case'' slogan. It distinguishes this parent-tree notion from the weaker monotone-order predicate currently named \texttt{TreeStructured}.

\leanmetapending{\LH{FR4}}
\begin{proposition}[Current Positive Interfaces Factor Through Role Classification]\label{prop:interfaces-factor-through-role}
The current positive theorem surface factors through the role classifier of Theorem~\ref{thm:current-landscape-partition}. Concretely, the family-indexed interfaces for
\begin{enumerate}
\item subcase-facing tractability,
\item degenerate-case witnesses, and
\item complexity-class summaries
\end{enumerate}
all factor through the map
\[
\text{family} \longmapsto \text{role} \in \{\text{core},\text{lifted},\text{degenerate}\}.
\]
\end{proposition}

Proposition~\ref{prop:interfaces-factor-through-role} shows that the exported positive API is already controlled by the role partition rather than by fifteen unrelated theorem schemas.
