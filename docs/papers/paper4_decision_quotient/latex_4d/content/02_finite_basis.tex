\section{Finite Basis for the Current Artifact}

We distinguish between \emph{formalized tractable families} and \emph{primitive tractability mechanisms}. A formalized family is any tractable condition with its own theorem statement or formal module in the current development. A primitive mechanism is a smaller conceptual source of tractability that may explain several families at once.

For the present artifact, the primitive list is explicit:
\begin{enumerate}
\item the six core structural subcases from the optimizer-quotient manuscript~\cite{simas2025optimizerquotient},
\item constant-optimizer collapse, and
\item finite explicit enumeration.
\end{enumerate}

Thus the current primitive basis has size eight.

\leanmetapending{\LH{FR1}, \LH{FR2}}
\begin{theorem}[Explicit Partition of the Current Formalized Landscape]\label{thm:current-landscape-partition}
The fifteen currently formalized tractable families admit an explicit disjoint partition into three blocks:
\begin{enumerate}
\item six \emph{core} structural families,
\item four \emph{lifted} families that reduce to existing core subcases, and
\item five \emph{degenerate} families that become tractable by optimizer collapse or finite explicit enumeration.
\end{enumerate}
Moreover, every currently formalized family is classified into exactly one of these three blocks by an explicit formal classification map.
\end{theorem}

\leanmetapending{\LH{FR25}, \LHrng{FR}{26}{28}}
\begin{proposition}[Explicit Inventory]\label{prop:explicit-inventory}
The fifteen currently formalized families are exactly the following:
\begin{enumerate}
\item core: bounded actions, separable utility, low tensor rank, tree structure, bounded treewidth, coordinate symmetry;
\item lifts: product distribution, bounded support, bounded horizon, full observability;
\item degenerate: single action, bounded state space, strict global dominance, constant optimal set, multiplicative-separable constant-sign.
\end{enumerate}
\end{proposition}

\leanmetapending{\LH{FR3}}
\begin{corollary}[Finite Basis for the Current Mechanized Landscape]\label{cor:finite-basis-current}
Projecting the partition of Theorem~\ref{thm:current-landscape-partition} to primitive mechanisms yields an explicit finite list of eight primitive tractability mechanisms. The list consists of the six core structural subcases
\[
\text{bounded actions},\ 
\text{separable utility},\ 
\text{low tensor rank},\ 
\text{tree structure},\ 
\text{bounded treewidth},\ 
\text{coordinate symmetry},
\]
together with two degenerate mechanisms:
\[
\text{constant-optimizer collapse}
\qquad\text{and}\qquad
\text{finite explicit enumeration}.
\]
\end{corollary}

\end{proposition}

\qquad

The current formalized landscape therefore contains fifteen tractable families accounted for by eight primitive mechanisms.

\leanmetapending{\LH{FR101}, \LH{FR105}, \LH{FR106}, \LH{FR108}, \LH{FR112}, \LH{FR113}}
\begin{proposition}[All Independent Core Mechanisms Are Nondegenerate]\label{prop:four-core-nondegenerate}
The current formal development already contains witness families showing that five core mechanisms are individually indispensable in the sense of basis coverage:
\begin{enumerate}
\item a low-rank-only witness family,
\item a separable-only witness family,
\item a bounded-actions-only witness family, and
\item a bounded-treewidth-only witness family, and
\item a symmetry-only witness family.
\end{enumerate}
Each of these families remains tractable by its designated mechanism while failing the currently formalized versions of the other independent core mechanisms.
\end{proposition}

Proposition~\ref{prop:four-core-nondegenerate} now settles the independent part of the basis question on the stable formal surface. Every core mechanism that is not merely contained in another is formally known to be genuinely needed.

More concretely, the current stable formal surface already contains five distinct kinds of ``only-by'' witness:
\begin{enumerate}
\item low-rank only,
\item separable only,
\item bounded-actions only, and
\item bounded-treewidth only, and
\item symmetry only.
\end{enumerate}
\leanmetapending{\LH{FR114}}
\begin{proposition}[Weak Tree Ordering Does Not Imply Width One]\label{prop:legacy-tree-gap}
There exists a dependency presentation satisfying the older monotone tree-order predicate \texttt{TreeStructured} whose dependency graph has real treewidth greater than one. In particular, width-one containment requires the stricter parent-tree condition from Proposition~\ref{prop:parent-tree-width-one}, not merely monotone ordering.
\end{proposition}

\leanmetapending{\LH{FR115}}
\begin{proposition}[Cyclic Dependencies Recover Hardness]\label{prop:tree-hardness-bridge}
The reduction family from the optimizer-quotient article already yields coNP-hard instances with cyclic dependency structure. Thus the tree-structured regime is not only sufficient for tractability; leaving it also reaches the hard side.
\end{proposition}

\leanmetapending{\LH{FR100}, \LH{FR110}, \LH{FR111}}
\begin{proposition}[Parent-Tree Structure Gives Width-One Decompositions]\label{prop:parent-tree-width-one}
Suppose a dependency presentation is tree-structured in the strict parent sense: every dependency points to a smaller coordinate and every coordinate has at most one parent. Then the induced dependency graph admits a genuine tree decomposition of width at most one.
\end{proposition}

Proposition~\ref{prop:parent-tree-width-one} is the mathematically honest containment theorem behind the intended ``tree structure is the treewidth-one special case'' slogan. The formal development distinguishes this parent-tree notion from the weaker monotone-order predicate currently named \texttt{TreeStructured} in the older tractability file.

\leanmetapending{\LH{FR4}}
\begin{proposition}[Current Positive Interfaces Factor Through Role Classification]\label{prop:interfaces-factor-through-role}
The current positive theorem surface factors through the role classifier of Theorem~\ref{thm:current-landscape-partition}. Concretely, the family-indexed interfaces for
\begin{enumerate}
\item subcase-facing tractability,
\item degenerate-case witnesses, and
\item complexity-class summaries
\end{enumerate}
all factor through the map
\[
\text{family} \longmapsto \text{role} \in \{\text{core},\text{lifted},\text{degenerate}\}.
\]
\end{proposition}

Proposition~\ref{prop:interfaces-factor-through-role} is the formal reason the current landscape feels smaller than its file count. The exported positive API is already controlled by the role partition rather than by fifteen unrelated theorem schemas.
