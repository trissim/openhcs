\section{Convergence of Rank, Tractability, and Cost}\label{five-way-equivalence}

Degree of freedom equals structural rank, and structural rank bounds decision entropy. The same rank-$1$ regime also appears as tractable sufficiency and minimum thermodynamic cost. An imported coherence theorem gives a separate single-source interpretation of that same point.

\subsection{Imported Coherence Reading}

\begin{theorem}[Coherent Single-Source Regime]\label{thm:coherent-single-source}
A bounded decision system lies in the coherent unit-independent-rate regime if and only if $\mathrm{DOF}(A)=1$.
\end{theorem}
\leanmeta{Imported coherence theorem (\LH{ORA1})}

\begin{informal}
In the imported coherence development, rank~$1$ means exactly one locus is authoritative, every remaining encoding is a derived view, and all reachable states remain coherent.
\end{informal}

\subsection{Structural Rank}

\begin{theorem}[Rank Identification]\label{thm:rank-identification}
For every bounded decision system $A$,
\[
\mathrm{srank}(\mathrm{canonicalDP}(A)) = \mathrm{DOF}(A).
\]
\end{theorem}
\leanmeta{\LH{L43}, \LH{L46}, \LH{L51}}

\begin{informal}
The degree-of-freedom count is not separate from the decision-theoretic object; it is exactly the interaction dimension of the canonical decision problem.
\end{informal}

\subsection{Tractability Boundary}

\begin{theorem}[Tractable Sufficiency at Rank One]\label{thm:tractable-rank-one}
In the canonical decision problem family, structural rank~$1$ is the tractable sufficiency regime, while higher structural rank enters the hard regime.
\end{theorem}
\leanmeta{Imported tractability theorem (\LH{L56}, \LH{L53}, \LH{L47})}

\begin{informal}
When exactly one relevant coordinate survives, exact sufficiency certification is tractable. Once more than one relevant coordinate survives, the canonical family crosses into the imported hardness regime.
\end{informal}

\subsection{Thermodynamic Selection}

\begin{theorem}[Thermodynamic Selection]\label{thm:thermodynamic-selection}
In the canonical decision encoding, every bounded decision system with $\mathrm{DOF}(A)>1$ lies strictly above the rank-$1$ Landauer ground state in per-cycle resolution cost.
\end{theorem}
\leanmeta{\LH{L49}, \LHrng{L}{54}{55}, \LH{BA8}}

\begin{informal}
The theorem uses only the rank identity together with Landauer calibration. Stronger hardness consequences require additional imported hypotheses.
\end{informal}

\subsection{Convergence Theorem}

\begin{theorem}[Convergence]\label{thm:five-way}
For every bounded decision system $A$, the following conditions are equivalent:
\[
\mathrm{DOF}(A)=1
\iff \mathrm{srank}(\mathrm{canonicalDP}(A))=1
\iff \text{tractable sufficiency for } \mathrm{canonicalDP}(A)
\iff \text{minimum per-cycle thermodynamic cost}.
\]
\end{theorem}
\leanmeta{\LHrng{L}{43}{44}, \LH{L47}, \LH{L55}}

\begin{informal}
The same rank-$1$ regime is simultaneously the exact one-coordinate regime, the tractable exact-certification regime, and the thermodynamic ground state. The imported coherence theorem gives an additional single-source interpretation of the same point.
\end{informal}

\textbf{Proof.}
\begin{enumerate}
\item Theorem~\ref{thm:rank-identification} identifies $\mathrm{DOF}(A)=1$ with structural rank~$1$.
\item Theorem~\ref{thm:tractable-rank-one} identifies structural rank~$1$ with the tractable sufficiency regime for the canonical family.
\item Theorem~\ref{thm:thermodynamic-selection} identifies rank~$1$ as the unique minimum-cost thermodynamic regime.
\end{enumerate}
Transitivity of logical equivalence completes the proof.

Theorem~\ref{thm:coherent-single-source} supplies an imported single-source interpretation of the same rank-$1$ regime.

\subsection{England Replication Inequality}

The England Replication Inequality is mechanized in \nolinkurl{Leverage/BridgeToDQ.lean} (\LH{L45}).

\begin{theorem}[England Replication Inequality]\label{thm:england}
Let $\Delta S_{\min}(r) = r \cdot k_B \ln 2$ be the rank-indexed minimal entropy production under Landauer calibration. For the rank-$1$ ground regime and any replicated rank-$k$ regime:
\[
\Delta S_{\min}(1) + k_B \ln k \leq \Delta S_{\min}(k)
\]
equivalently, $\Delta S_{\min}(k) - \Delta S_{\min}(1) \geq k_B \ln k$.
\end{theorem}

\leanmeta{\LH{L45}}

\begin{proof}
The gap is $(k-1) \cdot k_B \ln 2$. Since $k \leq 2^{k-1}$ (\LH{L52}), taking logs gives $\ln k \leq (k-1) \ln 2$, so the gap is $\geq k_B \ln k$.
\end{proof}

\textbf{Modeling note.} $\Delta S_{\min}$ is a definition within the model: the exact Landauer entropy cost of the canonical exact-resolution cycle. The ``min'' refers to physical optimality inside the calibrated decision model. In England's 2013 stochastic-thermodynamic framework \cite{england2013statistical}, the comparison point is the entropy premium associated with replication above a single-copy baseline. The present reformulation does not reproduce England's full path-space dynamics or detailed-balance setting. It isolates the same $k_B \ln k$ multiplicity penalty as a finite-counting consequence of rank-indexed Landauer cost together with $k \le 2^{k-1}$, making the selection penalty explicit in a finite exact-resolution model.

\subsection{Finite-Budget No-Collapse}

\begin{theorem}[Finite-Budget No-Collapse]\label{thm:finite-budget-no-collapse}
Let $B : \mathbb{N} \to \mathbb{N}$ be a budget profile, let $\mathrm{ops} : \mathbb{N} \to \mathbb{N}$ be a required-operation profile, and let $\mathrm{bitCost} > 0$ be the per-bit physical cost. If
\begin{enumerate}
\item $B$ is globally bounded,
\item $\mathrm{ops}$ has an exponential lower bound, and
\item collapse means: for every input size $n$, some feasible bit budget realizes at least $\mathrm{ops}(n)$ operations within budget $B(n)$,
\end{enumerate}
then no such physical collapse profile exists.
\end{theorem}
\leanmeta{\LH{PH26}}

\begin{proof}
This is exactly the bounded-budget physical no-collapse theorem in the physical-hardness layer. Exponential growth eventually exceeds every fixed finite budget, and positive per-bit cost lifts that growth into an energy contradiction.
\end{proof}

\begin{remark}[Interpretation]
Finite budget, positive event cost, and exponential exact-certification demand are jointly incompatible with a physical collapse model. Any stronger complexity-collapse conclusion requires an additional bridge from the chosen complexity claim to such a collapse profile.
\end{remark}

\begin{corollary}[$P = NP$ No-Go Transfer]\label{cor:pnp-nogo}
Assume a polynomial-collapse claim supplies a bridge to the finite-budget collapse profile of Theorem~\ref{thm:finite-budget-no-collapse}. Then that claim is physically impossible in the same model. In particular, any bridge from a $P = NP$ collapse claim to such a feasible physical-collapse profile yields a contradiction.
\end{corollary}
\leanmeta{\LHrng{PH}{14}{15}, \LH{PH26}}

\begin{proof}
Theorem~\ref{thm:finite-budget-no-collapse} rules out the collapse profile itself. The transfer theorem states that if a $P = NP$ claim implies that profile, then the claim is false in the model.
\end{proof}
