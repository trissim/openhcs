\section{Complexity Boundary of Exact Molecular Docking}\label{complexity-boundary}

Exact molecular docking has a genuine tractability boundary because the exact object already carries both qualitative and quantitative certification lower bounds. General exact sufficiency certification contains a hardness core. Sound checking requires witness budget. Molecular locality, sampling hypotheses, and concrete scorer approximations then carve out theorem-backed low-rank and stability regimes inside that harder ambient problem class. The claims in this section isolate that boundary before the thermodynamic lower bounds are applied.

\subsection{General Hardness Core}

\begin{theorem}[General Hardness Core for Exact Sufficiency]\label{thm:exact-sufficiency-hardness-core}
\leanmeta{\LH{L63}}
For every Boolean formula $\phi$, the empty coordinate set is sufficient for the reduction problem induced by $\phi$ if and only if $\phi$ is a tautology.
\end{theorem}

\begin{proof}
The reduction theorem is exact: tautology is encoded as sufficiency of the empty coordinate set for the induced decision problem. The equivalence therefore supplies the formal hardness core for exact sufficiency certification.
\end{proof}

\begin{theorem}[Maximal Structural Rank in the Hard Family]\label{thm:hard-family-srank}
\leanmeta{\LH{L64}}
Let $n>0$, and let $\phi$ be a non-tautology over $n$ Boolean variables. Then the many-coordinate reduction family has structural rank exactly $n$.
\end{theorem}

\begin{proof}
Every coordinate is relevant in the non-tautology branch of the strengthened reduction family. Structural rank is the cardinality of the relevant-coordinate set, so the rank is exactly $n$.
\end{proof}

The hard family therefore witnesses full interaction dimensionality: exact sufficiency can force the decision boundary to depend on every available coordinate.

\subsection{Quantitative Certification Lower Bounds}

\begin{theorem}[Checker Budget Lower Bound]\label{thm:checker-budget-lower-bound}
\leanmeta{\LH{L69}}
For the empty-set sufficiency core on $n \ge 1$ coordinates, any sound finite checker must inspect at least
\[
2^{n-1}
\]
pair witnesses.
\end{theorem}

\begin{proof}
The witness budget for the empty-set core is $2^{n-1}$. Any sound checker must inspect enough witness pairs to refute every false empty-set sufficiency claim, so the checking budget is bounded below by that witness budget.
\end{proof}

\begin{corollary}[No Sound Checker Below Witness Budget]\label{cor:no-sound-checker-below-budget}
\leanmeta{\LH{L71}}
For the same empty-set core, any checker operating strictly below the witness budget fails to be sound.
\end{corollary}

\begin{proof}
This is the contrapositive form of Theorem~\ref{thm:checker-budget-lower-bound}.
\end{proof}

\begin{corollary}[Checking Time Lower Bound]\label{cor:checking-time-lower-bound}
\leanmeta{\LH{L70}}
If runtime is bounded below by the number of checked witness pairs, then any sound checker for the empty-set core requires runtime at least
\[
2^{n-1}.
\]
\end{corollary}

\begin{proof}
The checking budget lower bound transfers directly to runtime once runtime dominates the number of checked pairs.
\end{proof}

The hardness core is therefore quantitative as well as qualitative: exact certification is expensive not only by reduction, but by unavoidable witness budget.

\subsection{Cutoff-Local Docking Regime}

\begin{theorem}[Cutoff-Local Structural-Rank Bound for Exact Docking]\label{thm:molecular-docking-srank-bound}
\leanmeta{\LH{L66}}
Let $P_{\mathrm{rel}}$ be the number of protein atoms within the cutoff radius of the binding site, and let $L$ be the number of ligand atoms. Under the strict-optimum and outside-cutoff boundedness hypotheses,
\[
\mathrm{srank}(D_{\mathrm{dock}}) \le 3P_{\mathrm{rel}} + 3L.
\]
\end{theorem}

\begin{proof}
Any protein coordinate that remains decision-relevant must come from an atom within the cutoff radius, while every ligand coordinate remains available. Structural rank is therefore bounded by three coordinates for each cutoff-local protein atom together with three coordinates for each ligand atom.
\end{proof}

\begin{corollary}[Bounded-Pocket Low-Rank Regime]\label{cor:bounded-pocket-regime}
\leanmeta{\LH{L65}}
If at most $K$ protein atoms lie within the cutoff radius of the binding site and the ligand has at most $L$ atoms, then
\[
\mathrm{srank}(D_{\mathrm{dock}}) \le 3K + 3L.
\]
\end{corollary}

\begin{proof}
Substitute the pocket-size and ligand-size bounds into Theorem~\ref{thm:molecular-docking-srank-bound}.
\end{proof}

Informally: bounded pockets bound exact difficulty.

\subsection{Sampled Docking}

\begin{theorem}[Sampled Exact-Coarse Winner Preservation]\label{thm:sampled-docking-gap}
\leanmeta{\LH{L67}}
For a finite sampled docking problem, let $a_\ast$ be a strict exact winner at a sampled state $s$. If the coarse score differs from the exact score by at most $\delta$ on every sampled action at $s$, and
\[
\delta < \frac{1}{2}\,\mathrm{StrictUtilityGap}(a_\ast,s),
\]
then the exact and coarse optimal-action sets agree at $s$.
\end{theorem}

\begin{proof}
The theorem is the sampled docking specialization of the strict half-gap invariance principle: a perturbation smaller than half the exact winner's strict margin cannot change the optimal set.
\end{proof}

\begin{theorem}[Inside-Cutoff Sufficiency for Sampled Docking]\label{thm:sampled-inside-cutoff-sufficient}
\leanmeta{\LH{L68}}
Under the cutoff boundedness, sampled-optimum capture, coordinate-compatibility, and injectivity hypotheses, the retained coordinate set consisting of inside-cutoff protein coordinates together with all ligand coordinates is sufficient for the sampled restricted docking problem.
\end{theorem}

\begin{proof}
Cutoff locality forces every relevant sampled coordinate into the retained set. The compatibility and injectivity hypotheses then lift the retained-set relevance bound into a sufficiency theorem for the sampled restricted problem.
\end{proof}

\subsection{Top-k and Near-Tie Control}

\begin{theorem}[Top-k Preservation Under Boundary Gap]\label{thm:topk-boundary-gap}
\leanmeta{\LH{L79}}
Let $u_{\mathrm{exact}}$ and $u_{\mathrm{coarse}}$ be finite score functions on a docking action family. If the coarse score differs from the exact score by at most $\delta$ on every action, and if $\delta$ is no larger than the boundary gap at threshold $\tau$, then every exact top-$k$ action survives the coarse threshold filter at $\tau$.
\end{theorem}

\begin{proof}
The boundary-gap condition places every exact top-$k$ action at least $\delta$ above the threshold. Uniform score error bounded by $\delta$ therefore keeps every exact top-$k$ action above the coarse threshold as well.
\end{proof}

\begin{theorem}[Exact Top-k Ambiguity-Band Containment]\label{thm:topk-ambiguity-band}
\leanmeta{\LH{L72}}
For every nonnegative slack parameter $\varepsilon$, every exact top-$k$ action lies inside the certified ambiguity band of width $\varepsilon$ around the exact $k$th boundary.
\end{theorem}

\begin{proof}
The ambiguity band is defined by lowering the exact $k$th threshold by $\varepsilon$. Every exact top-$k$ action remains above that relaxed threshold and is therefore retained.
\end{proof}

These top-$k$ theorems give a conservative exact-screening regime even when strict single-winner separation is unavailable.

\subsection{Concrete Scorer Families}

\begin{theorem}[Cutoff Lennard-Jones Winner Preservation]\label{thm:lj-cutoff-invariance}
\leanmeta{\LH{L75}}
For a finite sampled Lennard-Jones docking family, let $a_\ast$ be a strict exact winner at state $s$. If the finite cutoff error radius is smaller than half the strict exact utility gap at $s$, then the exact and cutoff Lennard-Jones optimal-action sets agree at $s$.
\end{theorem}

\begin{proof}
The finite cutoff radius gives a uniform approximation theorem for exact and cutoff Lennard-Jones scores on the sampled domain. A strict half-gap bound then forces winner preservation.
\end{proof}

\begin{theorem}[Cutoff Coulomb Uniform Approximation]\label{thm:coulomb-cutoff-uniform-approx}
\leanmeta{\LH{L74}}
For a finite sampled Coulomb docking family, the exact and cutoff Coulomb score families differ by at most the finite cutoff error radius uniformly over the sampled action-state domain.
\end{theorem}

\begin{proof}
The cutoff error radius is defined as the maximum exact-versus-cutoff discrepancy over the finite sampled domain. Uniform approximation follows immediately from that extremal definition.
\end{proof}

\begin{theorem}[Cutoff Coulomb Winner Preservation]\label{thm:coulomb-cutoff-invariance}
\leanmeta{\LH{L73}}
For a finite sampled Coulomb docking family, let $a_\ast$ be a strict exact winner at state $s$. If the finite cutoff error radius is smaller than half the strict exact utility gap at $s$, then the exact and cutoff Coulomb optimal-action sets agree at $s$.
\end{theorem}

\begin{proof}
The uniform cutoff-error bound from Theorem~\ref{thm:coulomb-cutoff-uniform-approx} combines with the strict half-gap criterion to force winner preservation.
\end{proof}
