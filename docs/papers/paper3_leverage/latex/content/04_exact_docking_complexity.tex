\section{Complexity Boundary of Exact Molecular Docking}\label{complexity-boundary}

Exact molecular docking has a genuine tractability boundary because the exact object already carries both qualitative and quantitative certification lower bounds. General exact sufficiency certification contains a hardness core. Sound checking requires witness budget. Molecular locality, sampling hypotheses, and concrete scorer approximations then carve out theorem-backed low-rank and stability regimes inside that harder ambient problem class. The claims in this section isolate that boundary before the thermodynamic lower bounds are applied.

\subsection{General Hardness Core}

\begin{theorem}[General Hardness Core for Exact Sufficiency]\label{thm:exact-sufficiency-hardness-core}
\leanmeta{\LH{L63}}
For every Boolean formula $\phi$, the empty coordinate set is sufficient for the reduction problem induced by $\phi$ if and only if $\phi$ is a tautology.
\end{theorem}

\begin{proof}
The reduction theorem is exact: tautology is encoded as sufficiency of the empty coordinate set for the induced decision problem. The equivalence therefore supplies the formal hardness core for exact sufficiency certification.
\end{proof}

\begin{theorem}[Maximal Structural Rank in the Hard Family]\label{thm:hard-family-srank}
\leanmeta{\LH{L64}}
Let $n>0$, and let $\phi$ be a non-tautology over $n$ Boolean variables. Then the many-coordinate reduction family has structural rank exactly $n$.
\end{theorem}

\begin{proof}
Every coordinate is relevant in the non-tautology branch of the strengthened reduction family. Structural rank is the cardinality of the relevant-coordinate set, so the rank is exactly $n$.
\end{proof}

The hard family therefore witnesses full interaction dimensionality: exact sufficiency can force the decision boundary to depend on every available coordinate.

\subsection{Quantitative Certification Lower Bounds}

\begin{theorem}[Checker Budget Lower Bound]\label{thm:checker-budget-lower-bound}
\leanmeta{\LH{L69}}
For the empty-set sufficiency core on $n \ge 1$ coordinates, any sound finite checker must inspect at least
\[
2^{n-1}
\]
pair witnesses.
\end{theorem}

\begin{proof}
The witness budget for the empty-set core is $2^{n-1}$. Any sound checker must inspect enough witness pairs to refute every false empty-set sufficiency claim, so the checking budget is bounded below by that witness budget.
\end{proof}

\begin{corollary}[No Sound Checker Below Witness Budget]\label{cor:no-sound-checker-below-budget}
\leanmeta{\LH{L71}}
For the same empty-set core, any checker operating strictly below the witness budget fails to be sound.
\end{corollary}

\begin{proof}
This is the contrapositive form of Theorem~\ref{thm:checker-budget-lower-bound}.
\end{proof}

\begin{corollary}[Checking Time Lower Bound]\label{cor:checking-time-lower-bound}
\leanmeta{\LH{L70}}
If runtime is bounded below by the number of checked witness pairs, then any sound checker for the empty-set core requires runtime at least
\[
2^{n-1}.
\]
\end{corollary}

\begin{proof}
The checking budget lower bound transfers directly to runtime once runtime dominates the number of checked pairs.
\end{proof}

The hardness core is therefore quantitative as well as qualitative: exact certification carries an unavoidable witness budget in addition to the reduction-theoretic hardness statement.

\subsection{Cutoff-Local Docking Regime}

\begin{theorem}[Cutoff-Local Structural-Rank Bound for Exact Docking]\label{thm:molecular-docking-srank-bound}
\leanmeta{\LH{L66}}
Let $P_{\mathrm{rel}}$ be the number of protein atoms within the cutoff radius of the binding site, and let $L$ be the number of ligand atoms. Under the strict-optimum and outside-cutoff boundedness hypotheses,
\[
\mathrm{srank}(D_{\mathrm{dock}}) \le 3P_{\mathrm{rel}} + 3L.
\]
\end{theorem}

\begin{proof}
Any protein coordinate that remains decision-relevant must come from an atom within the cutoff radius, while every ligand coordinate remains available. Structural rank is therefore bounded by three coordinates for each cutoff-local protein atom together with three coordinates for each ligand atom.
\end{proof}

\begin{corollary}[Bounded-Pocket Low-Rank Regime]\label{cor:bounded-pocket-regime}
\leanmeta{\LH{L65}}
If at most $K$ protein atoms lie within the cutoff radius of the binding site and the ligand has at most $L$ atoms, then
\[
\mathrm{srank}(D_{\mathrm{dock}}) \le 3K + 3L.
\]
\end{corollary}

\begin{proof}
Substitute the pocket-size and ligand-size bounds into Theorem~\ref{thm:molecular-docking-srank-bound}.
\end{proof}

Informally: bounded pockets bound exact difficulty.

\subsection{Conformer-Search Collapse Regimes}

\begin{theorem}[Strict Dominance Makes Coordinates Irrelevant]\label{thm:strict-dominance-empty-sufficient}
\leanmeta{\LH{L85}}
Let $D$ be a finite exact docking problem. If one action strictly dominates every other action at every state, then every coordinate set is sufficient for $D$.
\end{theorem}

\begin{proof}
Strict global dominance makes the dominant action uniquely optimal at every state. The optimal-action set is therefore constant, so no coordinate distinction can change exact resolution.
\end{proof}

\begin{theorem}[Multiplicative Separability Eliminates Conformer Dependence]\label{thm:multiplicative-separable-empty-sufficient}
\leanmeta{\LH{L84}}
Let $D$ be a finite exact docking problem with utility of the form
\[
U(a,s)=f(a)g(s).
\]
If the state factor $g$ has constant sign, then every coordinate set is sufficient for $D$.
\end{theorem}

\begin{proof}
With constant-sign state factor, the optimal-action set depends only on the ordering of the action factor $f$. The optimizer is therefore independent of state and exact sufficiency collapses to the empty-information regime.
\end{proof}

These theorems give exact collapse regimes for conformer search: under strict dominance or constant-sign multiplicative separability, conformational variation does not change the exact optimizer.

\subsection{Sampled Docking}

\begin{theorem}[Sampled Exact-Coarse Winner Preservation]\label{thm:sampled-docking-gap}
\leanmeta{\LH{L67}}
For a finite sampled docking problem, let $a_\ast$ be a strict exact winner at a sampled state $s$. If the coarse score differs from the exact score by at most $\delta$ on every sampled action at $s$, and
\[
\delta < \frac{1}{2}\,\mathrm{StrictUtilityGap}(a_\ast,s),
\]
then the exact and coarse optimal-action sets agree at $s$.
\end{theorem}

\begin{proof}
The theorem is the sampled docking specialization of the strict half-gap invariance principle: a perturbation smaller than half the exact winner's strict margin cannot change the optimal set.
\end{proof}

\begin{theorem}[Action-Pinned Sampled Docking Has a Uniform Positive Strict Gap]\label{thm:action-pinned-uniform-gap}
\leanmeta{\LHrng{L}{115}{117}}
Let
\[
  D^{\mathrm{flat}}_{\mathrm{dock,pin}}
\]
be the action-side symmetry-broken refinement of the flat exact sampled docking problem, and let $a^\dagger(s)$ denote the canonical max-code representative of the original exact optimal set at flat sampled grid state $s$. If the sampled action family has more than one retained action, then there exists $\gamma>0$ such that for every flat sampled grid state $s$,
\[
\gamma \le \mathrm{StrictUtilityGap}_{D^{\mathrm{flat}}_{\mathrm{dock,pin}}}(a^\dagger(s),s).
\]
Moreover,
\[
\mathrm{srank}\!\bigl(D^{\mathrm{flat}}_{\mathrm{dock,pin}}\bigr)
\le
\mathrm{srank}\!\bigl(D^{\mathrm{flat}}_{\mathrm{dock,exact}}\bigr).
\]
\end{theorem}

\begin{proof}
Action-side pinning canonically chooses a strict sampled winner from each exact tie class. The resulting strict utility gap is positive at every flat sampled grid state, and finiteness of the sampled grid state space yields a positive global minimum $\gamma$. Because the symmetry breaking acts only on the action side, it does not increase the number of state coordinates needed for exact resolution.
\end{proof}

This action-pinned refinement is the theorem-backed replacement for a generic no-ties assumption. It preserves the state-side exact-resolution problem, does not increase structural rank, and converts every finite sampled tie class into a canonical strict winner with a uniform positive margin. The later half-gap invariance theorems can therefore be applied to the pinned sampled family with an explicit witness $\gamma$ rather than an ad hoc generic-position premise.

\begin{theorem}[Inside-Cutoff Sufficiency for Sampled Docking]\label{thm:sampled-inside-cutoff-sufficient}
\leanmeta{\LH{L68}}
Under the cutoff boundedness, sampled-optimum capture, coordinate-compatibility, and injectivity hypotheses, the retained coordinate set consisting of inside-cutoff protein coordinates together with all ligand coordinates is sufficient for the sampled restricted docking problem.
\end{theorem}

\begin{proof}
Cutoff locality forces every relevant sampled coordinate into the retained set. The compatibility and injectivity hypotheses then lift the retained-set relevance bound into a sufficiency theorem for the sampled restricted problem.
\end{proof}

\subsection{Top-k and Near-Tie Control}

\begin{theorem}[Top-k Preservation Under Boundary Gap]\label{thm:topk-boundary-gap}
\leanmeta{\LH{L79}}
Let $u_{\mathrm{exact}}$ and $u_{\mathrm{coarse}}$ be finite score functions on a docking action family. If the coarse score differs from the exact score by at most $\delta$ on every action, and if $\delta$ is no larger than the boundary gap at threshold $\tau$, then every exact top-$k$ action survives the coarse threshold filter at $\tau$.
\end{theorem}

\begin{proof}
The boundary-gap condition places every exact top-$k$ action at least $\delta$ above the threshold. Uniform score error bounded by $\delta$ therefore keeps every exact top-$k$ action above the coarse threshold as well.
\end{proof}

\begin{theorem}[Exact Top-k Ambiguity-Band Containment]\label{thm:topk-ambiguity-band}
\leanmeta{\LH{L72}}
For every nonnegative slack parameter $\varepsilon$, every exact top-$k$ action lies inside the certified ambiguity band of width $\varepsilon$ around the exact $k$th boundary.
\end{theorem}

\begin{proof}
The ambiguity band is defined by lowering the exact $k$th threshold by $\varepsilon$. Every exact top-$k$ action remains above that relaxed threshold and is therefore retained.
\end{proof}

These top-$k$ theorems give a conservative exact-screening regime even when strict single-winner separation is unavailable.

\subsection{Continuous-to-Discrete Approximation Bridges}

\begin{theorem}[Lipschitz Grid Error Gives Resolution-Controlled Approximation]\label{thm:lipschitz-grid-approx}
\leanmeta{\LH{L90}}
Let $u_{\mathrm{cont}}$ be a continuous-state score family, let $u_{\mathrm{grid}}$ be its grid approximation, and let each grid state carry a state-discretization error bounded by $\mathrm{res}$. If the score discrepancy is Lipschitz with constant $L$, then the exact-versus-grid score error is uniformly bounded by $L\,\mathrm{res}$.
\end{theorem}

\begin{proof}
The Lipschitz hypothesis bounds score discrepancy at each grid state by $L$ times its state error. A uniform state-error bound by $\mathrm{res}$ therefore yields a uniform score radius $L\,\mathrm{res}$.
\end{proof}

\begin{theorem}[Resolution-Controlled Approximation Gives Uniform Utility Approximation]\label{thm:resolution-controlled-uniform}
\leanmeta{\LH{L91}}
If the exact-versus-grid score discrepancy is bounded by a resolution-dependent envelope $\varepsilon(\mathrm{res})$ on every lifted grid state, then the lifted grid problem is a uniform utility approximation with radius $\varepsilon(\mathrm{res})$.
\end{theorem}

\begin{proof}
The resolution-controlled hypothesis already states the required pointwise uniform score bound on the lifted grid state space.
\end{proof}

\begin{theorem}[Resolution-Controlled Approximation Preserves Exact Winners]\label{thm:resolution-controlled-gap-invariance}
\leanmeta{\LH{L99}}
Let $u_{\mathrm{cont}}$ be a continuous score family, $u_{\mathrm{grid}}$ its discretized score family, and $\mathrm{lift}$ the map from grid states to continuous states. If the exact-versus-grid score discrepancy is bounded by $\varepsilon(\mathrm{res})$ on every lifted grid state, and if
\[
\varepsilon(\mathrm{res}) < \frac{1}{2}\,\mathrm{StrictUtilityGap},
\]
then the lifted exact optimizer and the grid optimizer agree at that grid state.
\end{theorem}

\begin{proof}
Resolution-controlled approximation yields a uniform utility approximation with radius $\varepsilon(\mathrm{res})$. The standard strict half-gap criterion then gives exact/coarse winner preservation.
\end{proof}

\begin{theorem}[Large Cutoff Implies Docking Cutoff Boundedness]\label{thm:large-cutoff-bounded}
\leanmeta{\LH{L89}}
Let a binding problem admit a lattice-tail perturbation bound with coefficient $c$ and a positive strict minimum decision gap $\gamma$. If the chosen cutoff radius satisfies
\[
c\,\mathrm{TailSum}_6(R_{\mathrm{cut}}) < \gamma/2,
\]
then the exact docking problem satisfies the cutoff-boundedness hypothesis required by the structural-rank and sampled-docking locality theorems.
\end{theorem}

\begin{proof}
The lattice-tail bound supplies a uniform upper bound on every outside-cutoff utility perturbation. The strict minimum gap condition then shows that the perturbation remains below half the exact decision gap, which is exactly the cutoff-boundedness criterion.
\end{proof}

These theorems provide a direct route from continuous regularity and physical tail control to the finite exact/coarse approximation theorems used later in the docking development.

\begin{theorem}[Explicit Lennard-Jones Shell Derivative Envelope]\label{thm:lj-shell-derivative-envelope}
\leanmeta{\LH{L103}}
Fix a positive lower shell radius $r_{\min}>0$. Then every radius $r \ge r_{\min}$ satisfies
\[
\left|\frac{d}{dr}U_{\mathrm{LJ}}(r)\right|
\le
\left(24|\varepsilon|r_{\min}^{-1}\right)
\left((|\sigma|r_{\min}^{-1})^6 + 2(|\sigma|r_{\min}^{-1})^{12}\right).
\]
\end{theorem}

\begin{proof}
The closed-form Lennard-Jones derivative is bounded by replacing every inverse radius factor with the lower-shell inverse $r_{\min}^{-1}$ and then bounding the polynomial terms by monotonicity on the positive shell.
\end{proof}

\begin{theorem}[Explicit Lennard-Jones Shell Second-Derivative Envelope]\label{thm:lj-shell-second-derivative-envelope}
\leanmeta{\LH{L105}}
Fix a positive lower shell radius $r_{\min}>0$. Then every radius $r \ge r_{\min}$ satisfies
\[
\left|\frac{d^2}{dr^2}U_{\mathrm{LJ}}(r)\right|
\le
\left(24|\varepsilon|r_{\min}^{-2}\right)
\left(26(|\sigma|r_{\min}^{-1})^{12} + 7(|\sigma|r_{\min}^{-1})^6\right).
\]
\end{theorem}

\begin{proof}
Differentiate the closed-form gradient once more and bound every inverse radius factor by the lower-shell inverse. The polynomial terms are again controlled by monotonicity on the positive shell.
\end{proof}

\begin{theorem}[Lennard-Jones Hessian Envelope Makes the Gradient Lipschitz]\label{thm:lj-shell-hessian-lipschitz}
\leanmeta{\LH{L106}}
On a positive shell $[r_{\min},r_{\max}]$, any uniform bound on the Lennard-Jones second derivative yields a Lipschitz bound on the Lennard-Jones gradient:
\[
|U'_{\mathrm{LJ}}(y)-U'_{\mathrm{LJ}}(x)| \le K|y-x|.
\]
\end{theorem}

\begin{proof}
The second derivative is the derivative of the gradient. A uniform shell bound on that derivative therefore makes the gradient Lipschitz by the one-dimensional mean value theorem.
\end{proof}

\begin{theorem}[Lennard-Jones Hessian Envelope Gives a Quadratic Taylor Remainder]\label{thm:lj-shell-quadratic-remainder}
\leanmeta{\LH{L108}}
Fix a positive shell $[r_{\min}, r_{\max}]$. Then for every $x,y \in [r_{\min}, r_{\max}]$,
\[
|U_{\mathrm{LJ}}(y)-U_{\mathrm{LJ}}(x)-U'_{\mathrm{LJ}}(x)(y-x)|
\le
\frac{1}{2}
\left(24|\varepsilon|r_{\min}^{-2}\right)
\left(26(|\sigma|r_{\min}^{-1})^{12} + 7(|\sigma|r_{\min}^{-1})^6\right)
|y-x|^2.
\]
\end{theorem}

\begin{proof}
The shell second-derivative envelope gives a uniform Hessian bound on the positive shell. The one-dimensional second-order Taylor remainder is therefore bounded by one half of that shellwise Hessian envelope times the squared radial displacement.
\end{proof}

\begin{theorem}[Lennard-Jones Shell Gradient Bound Gives Score Stability]\label{thm:lj-shell-grad-stability}
\leanmeta{\LH{L102}}
Fix a positive radial shell $[r_{\min}, r_{\max}]$ and assume the Lennard-Jones gradient magnitude is bounded by $C$ on that shell. Then for any two radii $x,y \in [r_{\min}, r_{\max}]$,
\[
|U_{\mathrm{LJ}}(y)-U_{\mathrm{LJ}}(x)| \le C|y-x|.
\]
\end{theorem}

\begin{proof}
The closed-form Lennard-Jones derivative is bounded by hypothesis on the positive shell. The one-dimensional mean value theorem then bounds variation of the potential by the derivative envelope times the radial displacement.
\end{proof}

\begin{theorem}[Lennard-Jones Shell Gradient Bound Gives Uniform Exact/Grid Approximation]\label{thm:lj-shell-uniform-approx}
\leanmeta{\LH{L101}}
Let $d_{\mathrm{exact}}$ and $d_{\mathrm{grid}}$ be exact and discretized distance maps whose values remain inside a positive shell $[r_{\min}, r_{\max}]$. If the Lennard-Jones gradient magnitude is bounded by $C$ on that shell and
\[
|d_{\mathrm{exact}}-d_{\mathrm{grid}}| \le \delta
\]
pointwise, then the exact and grid Lennard-Jones score families form a uniform utility approximation with radius $C\delta$.
\end{theorem}

\begin{proof}

\begin{theorem}[First-Order-Corrected Lennard-Jones Discretization Error Is Quadratic]\label{thm:lj-shell-quadratic-discretization}
\leanmeta{\LH{L107}}
Let $d_{\mathrm{exact}}$ and $d_{\mathrm{grid}}$ be exact and discretized distance maps whose values remain inside a positive shell $[r_{\min}, r_{\max}]$, and assume
\[
|d_{\mathrm{exact}}-d_{\mathrm{grid}}| \le \delta

pointwise. Then the first-order-corrected exact/grid Lennard-Jones discrepancy satisfies the quadratic bound
\[

\left(U_{\mathrm{LJ}}(d_{\mathrm{grid}})+U'_{\mathrm{LJ}}(d_{\mathrm{grid}})(d_{\mathrm{exact}}-d_{\mathrm{grid}})\right)
\right|
\le
\frac{1}{2}
\left(24|\varepsilon|r_{\min}^{-2}\right)
\left(26(|\sigma|r_{\min}^{-1})^{12} + 7(|\sigma|r_{\min}^{-1})^6\right)
\delta^2.
\]
\end{theorem}

\begin{proof}
Apply the quadratic Taylor remainder theorem to the shellwise exact-versus-grid radial displacement and then substitute the uniform distance-discretization bound $|d_{\mathrm{exact}}-d_{\mathrm{grid}}| \le \delta$.
\end{proof}
